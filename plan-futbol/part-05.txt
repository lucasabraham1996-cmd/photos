function boot(){
  const d=new Date(); const iso=d.toISOString().slice(0,10);
  document.getElementById('sessionDate').value=iso; document.getElementById('progressDate').value=iso;
  selectedDay=dayFromDate(d);
  if(window.google && google.script && google.script.run){
    google.script.run.withSuccessHandler(init).withFailureHandler(fail).getAppData();
  }else{
    DATA={config:{NombreApp:'Plan Fútbol & Definición · 45 min',ObjetivoPrincipal:'Bajar grasa/cintura + ganar brazos + mejorar fútbol + fortalecer gemelos',AlertaGemelo:'Dolor agudo/tirón: detené la carga.'},plan:DEMO_PLAN,tutorials:DEMO_TUTORIALS,recentSessions:[],recentProgress:[]};
    document.getElementById('standaloneMsg').style.display='block'; init(DATA);
  }
}
function dayFromDate(d){ const map={1:'Lunes',2:'Martes',3:'Miércoles',4:'Jueves',5:'Viernes'}; return map[d.getDay()]||'Lunes'; }
function init(data){
  DATA=data||DATA;
  document.getElementById('loading').style.display='none'; document.getElementById('main').style.display='block';
  document.getElementById('appTitle').textContent=DATA.config.NombreApp||'Plan Fútbol & Definición';
  document.getElementById('appObjective').textContent=DATA.config.ObjetivoPrincipal||'';
  document.getElementById('calfRule').textContent='🦵 '+(DATA.config.AlertaGemelo||'Gemelos: progresión gradual y sin dolor agudo.');
  document.getElementById('todayLabel').textContent='Hoy: '+dayFromDate(new Date());
  buildDays(); renderDay(); renderTutorialLibrary(); renderHistory(); renderProgressHistory();
}
function fail(err){ document.getElementById('loading').innerHTML='<div class="danger">No pude cargar la hoja.</div><p>'+escapeHtml(err.message||String(err))+'</p>'; }
function buildDays(){
  const root=document.getElementById('days'); root.innerHTML='';
  DAY_ORDER.forEach(day=>{ const b=document.createElement('button'); b.className='day-btn'+(day===selectedDay?' active':''); b.textContent=day; b.onclick=()=>{selectedDay=day;doneSet.clear();buildDays();renderDay();}; root.appendChild(b); });
}
function renderDay(){
  const items=DATA.plan.filter(x=>x.day===selectedDay).sort((a,b)=>a.order-b.order);
  const isRest=selectedDay==='Martes';
  document.getElementById('dayFocus').textContent=DAY_FOCUS[selectedDay]||selectedDay;
  document.getElementById('dayCount').textContent=isRest
    ? 'Descanso programado · sin ejercicios'
    : (selectedDay==='Viernes' ? items.length+' bloques · el partido cuenta como sesión' : items.length+' bloques · objetivo '+(DAY_DURATION[selectedDay]||'45 min'));
  document.getElementById('dayTag').textContent=isRest?'😴 Descanso':selectedDay==='Viernes'?'⚽ Fútbol':selectedDay==='Jueves'?'⚡ 45 min · ligero':'⏱️ 45 min';

  const form=document.getElementById('sessionFormCard');
  if(form) form.style.display=isRest?'none':'block';
  const minutes=document.getElementById('minutes');
  if(minutes) minutes.value=isRest?'0':'45';

  const root=document.getElementById('exerciseList'); root.innerHTML='';
  if(isRest){
    root.innerHTML=`<div class="rest-card"><span class="rest-emoji">😴</span><h3>Martes = descanso</h3><p>No hay rutina hoy. La recuperación es parte del plan: ayuda a que brazos, piernas y gemelos asimilen el trabajo y te permite rendir mejor el resto de la semana.</p><small>Si te sentís rígido, podés caminar muy suave 10-20 min por gusto, pero no es obligatorio ni cuenta como entrenamiento.</small></div>`;
    document.getElementById('dayProgress').style.width='100%';
    document.getElementById('dayProgressCaption').textContent='Descanso programado ✓';
    return;
  }
  if(!items.length){root.innerHTML='<div class="empty">No hay ejercicios cargados para este día.</div>';updateProgress(items);return;}
  items.forEach((x,i)=>{
    const id=selectedDay+'-'+x.order+'-'+x.exercise;
    const el=document.createElement('article'); el.className='exercise'+(doneSet.has(id)?' done':'');
    const warn=/dolor|tirón|SOLO|pinchazo/i.test(x.note||'');
    el.innerHTML=`<button class="check" aria-label="Completar">${doneSet.has(id)?'✓':''}</button><div><div class="block">${escapeHtml(x.block||'')}</div><div class="exercise-name">${escapeHtml(x.exercise||'')}</div><div class="exercise-meta"><span class="pill">${escapeHtml(x.sets||'-')} series</span><span class="pill">${escapeHtml(x.reps||'')}</span><span class="pill">${escapeHtml(x.equipment||'')}</span>${x.rest&&x.rest!=='-'?`<span class="pill">Desc. ${escapeHtml(x.rest)}</span>`:''}</div><div class="note">${escapeHtml(x.intensity||'')} · ${escapeHtml(x.objective||'')}</div>${x.note?`<div class="note ${warn?'warning-note':''}">${escapeHtml(x.note)}</div>`:''}<button class="learn-btn" type="button" onclick="openTutorialByEncoded('${encodeURIComponent(x.exercise||'')}')">🎥 Cómo hacerlo</button></div>`;
    el.querySelector('.check').onclick=()=>{ if(doneSet.has(id))doneSet.delete(id);else doneSet.add(id); renderDay(); };
    root.appendChild(el);
  });
  updateProgress(items);
}
function updateProgress(items){ const n=items.length, done=doneSet.size, p=n?Math.round(done/n*100):0; document.getElementById('dayProgress').style.width=p+'%'; document.getElementById('dayProgressCaption').textContent=done+'/'+n+' · '+p+'%'; }
function saveSession(){
  const items=DATA.plan.filter(x=>x.day===selectedDay);
  const payload={date:val('sessionDate'),day:selectedDay,session:DAY_FOCUS[selectedDay]||selectedDay,completed:doneSet.size,total:items.length,rpe:+val('rpe'),calfPain:+val('calfPain'),minutes:+val('minutes'),notes:val('sessionNotes')};
  if(!(window.google&&google.script&&google.script.run)){toast('Vista local: no se puede guardar en Google Sheets.');return;}
  setBusy('saveSessionBtn',true,'Guardando…');
  google.script.run.withSuccessHandler(()=>{toast('✅ Sesión guardada');doneSet.clear();setBusy('saveSessionBtn',false,'Guardar sesión en la hoja');refresh();}).withFailureHandler(e=>{toast('Error: '+e.message);setBusy('saveSessionBtn',false,'Guardar sesión en la hoja');}).saveSession(payload);
}
