function norm(v){return String(v||'').normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase().trim();}
function findTutorial(name){
  const n=norm(name);
  return (DATA.tutorials||[]).find(t=>norm(t.exercise)===n) || null;
}
function openTutorialByEncoded(encoded){openTutorial(decodeURIComponent(encoded));}
function openTutorial(name){
  const t=findTutorial(name);
  if(!t){toast('Todavía no hay tutorial cargado para este ejercicio.');return;}
  const p=(DATA.plan||[]).find(x=>norm(x.exercise)===norm(name));
  document.getElementById('tutorialTitle').textContent=t.exercise||name;
  document.getElementById('tutorialMeta').innerHTML=[p?.equipment,p?.sets?`${p.sets} series`:null,p?.reps,p?.rest&&p.rest!=='-'?`Desc. ${p.rest}`:null].filter(Boolean).map(v=>`<span class="pill">${escapeHtml(v)}</span>`).join('');

function closeTutorial(){
  const modal=document.getElementById('tutorialModal');
  modal.classList.remove('open'); modal.setAttribute('aria-hidden','true'); document.body.classList.remove('modal-open');
  if(tutorialAnimFrame){cancelAnimationFrame(tutorialAnimFrame);tutorialAnimFrame=null;}
}
function renderTutorialLibrary(filter=''){
  const root=document.getElementById('tutorialLibrary'); if(!root)return;
  const q=norm(filter);
  const seen=new Set();
  const ordered=[];
  (DATA.plan||[]).forEach(p=>{const k=norm(p.exercise);if(!seen.has(k)){seen.add(k);ordered.push(p);}});
  const items=ordered.filter(p=>!q || norm(p.exercise+' '+p.objective+' '+p.equipment).includes(q));
  if(!items.length){root.innerHTML='<div class="empty">No encontré ejercicios con ese texto.</div>';return;}
  root.innerHTML=items.map(p=>`<div class="guide-item"><div class="guide-item-main"><b>${escapeHtml(p.exercise)}</b><small>${escapeHtml(p.objective||'')} · ${escapeHtml(p.equipment||'')}</small></div><button class="guide-open" type="button" onclick="openTutorialByEncoded('${encodeURIComponent(p.exercise||'')}')">🎥 Ver</button></div>`).join('');
}

function P(x,y){return{x,y};}
function mix(a,b,t){return P(a.x+(b.x-a.x)*t,a.y+(b.y-a.y)*t);}
function clonePose(p){const o={};Object.keys(p).forEach(k=>o[k]=P(p[k].x,p[k].y));return o;}
function baseStanding(){return{head:P(160,32),ls:P(145,60),rs:P(175,60),le:P(140,92),re:P(180,92),lh:P(138,126),rh:P(182,126),lhip:P(150,116),rhip:P(170,116),lk:P(150,162),rk:P(170,162),la:P(146,205),ra:P(174,205)};}
function shiftPose(p,dx,dy){const o={};Object.keys(p).forEach(k=>o[k]=P(p[k].x+dx,p[k].y+dy));return o;}
function poseStanding(type,end){
  let a=baseStanding(),b=clonePose(a);
  if(type==='curl'||type==='hammercurl'){b.lh=P(138,76);b.rh=P(182,76);b.le=P(140,94);b.re=P(180,94);}
  else if(type==='shoulderpress'){a.le=P(130,88);a.re=P(190,88);a.lh=P(138,66);a.rh=P(182,66);b.le=P(148,46);b.re=P(172,46);b.lh=P(150,20);b.rh=P(170,20);}
  else if(type==='overheadtriceps'){a.le=P(145,54);a.re=P(175,54);a.lh=P(156,92);a.rh=P(164,92);b.le=P(145,48);b.re=P(175,48);b.lh=P(150,20);b.rh=P(170,20);}
  else if(type==='lateralraise'){a.lh=P(138,125);a.rh=P(182,125);b.le=P(115,66);b.re=P(205,66);b.lh=P(88,66);b.rh=P(232,66);}
  else if(type==='squat'){a.lh=P(148,92);a.rh=P(172,92);b.head=P(160,61);b.ls=P(145,88);b.rs=P(175,88);b.le=P(140,105);b.re=P(180,105);b.lh=P(148,100);b.rh=P(172,100);b.lhip=P(150,138);b.rhip=P(170,138);b.lk=P(125,168);b.rk=P(195,168);b.la=P(132,205);b.ra=P(188,205);}
  else if(type==='calfraise'){b=shiftPose(a,0,-10);b.la=P(146,199);b.ra=P(174,199);}
  else if(type==='bentcalf'){a.lk=P(140,170);a.rk=P(180,170);a.lhip=P(150,125);a.rhip=P(170,125);b=shiftPose(a,0,-8);b.la=P(146,200);b.ra=P(174,200);}
  else if(type==='tibialis'){/* body static; foot is animated in extras */}
  else if(type==='bandrow'){a.le=P(125,88);a.re=P(195,88);a.lh=P(95,92);a.rh=P(225,92);b.le=P(125,88);b.re=P(195,88);b.lh=P(145,94);b.rh=P(175,94);}
  else if(type==='bandtriceps'){a.le=P(145,84);a.re=P(175,84);a.lh=P(145,112);a.rh=P(175,112);b.le=P(145,84);b.re=P(175,84);b.lh=P(145,138);b.rh=P(175,138);}
  else if(type==='mobility'){b.lk=P(137,137);b.la=P(112,126);b.lh=P(130,95);b.rh=P(190,75);}
  else if(type==='lateralwalk'){a.lk=P(142,168);a.rk=P(178,168);a.la=P(132,205);a.ra=P(188,205);b=shiftPose(a,22,0);b.la.x-=8;b.ra.x+=8;b.lk.x-=5;b.rk.x+=5;}
  else {b.lh=P(140,78);b.rh=P(180,78);}
  return end?b:a;
}
